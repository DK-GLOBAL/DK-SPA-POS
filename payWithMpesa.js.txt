async function payWithMpesa(phone, amount, ref) {
  const { data: { session } } = await supabase.auth.getSession();
  const r = await fetch('/api/stkpush', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Authorization: 'Bearer ' + session.access_token },
    body: JSON.stringify({ phone, amount, ref })
  });
  const j = await r.json();
  if (!r.ok) throw new Error(j.error);
  for (let i = 0; i < 20; i++) {
    await new Promise((s) => setTimeout(s, 3000));
    const { data } = await supabase.from('mpesa_payments')
      .select('status,mpesa_receipt,result_desc')
      .eq('checkout_request_id', j.checkoutRequestId).single();
    if (data && data.status !== 'pending') return data; // 'paid' or 'failed'
  }
  return { status: 'pending' };
}
